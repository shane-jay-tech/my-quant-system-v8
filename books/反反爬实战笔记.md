## 二、延时与随机间隔

```python
import time
import random

# 基础随机延时
def random_delay(min_s=1.0, max_s=3.0):
    t = random.uniform(min_s, max_s)
    time.sleep(t)
    return t

# 令牌桶算法（精确控制 QPS）
class TokenBucket:
    def __init__(self, rate, capacity):
        self.rate = rate          # 每秒生成令牌数
        self.capacity = capacity  # 桶容量
        self.tokens = capacity
        self.last_time = time.time()

    def consume(self, tokens=1):
        now = time.time()
        self.tokens = min(self.capacity,
                          self.tokens + self.rate * (now - self.last_time))
        self.last_time = now
        if self.tokens >= tokens:
            self.tokens -= tokens
            return True
        return False

# 使用示例
bucket = TokenBucket(rate=2, capacity=5)  # 每秒最多2请求，突发5请求
for url in urls:
    while not bucket.consume():
        time.sleep(0.1)
    fetch(url)
```

### 黄金法则

- 页面列表请求：`random.uniform(2, 5)` 秒
- 详情页请求：`random.uniform(1, 3)` 秒
- API 接口请求：`random.uniform(0.5, 1.5)` 秒
- 被封后冷却：`random.uniform(30, 60)` 秒后重试

## 三、自动重试与指数退避

```python
import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

def create_session_with_retry():
    session = requests.Session()

    # 方式1：urllib3 内置重试
    retry_strategy = Retry(
        total=3,                      # 最多重试3次
        backoff_factor=1.5,           # 退避因子：1.5s, 3s, 6s
        status_forcelist=[429, 500, 502, 503, 504],
        allowed_methods=["GET", "POST"]
    )
    adapter = HTTPAdapter(max_retries=retry_strategy)
    session.mount("http://", adapter)
    session.mount("https://", adapter)
    return session

# 方式2：手动指数退避（更灵活）
def fetch_with_backoff(url, max_retries=3):
    session = create_session_with_retry()
    for attempt in range(max_retries):
        try:
            response = session.get(url, timeout=30)
            if response.status_code == 200:
                return response
            if response.status_code == 403:
                print(f"[WARN] 403 Forbidden, switching source...")
                return None
        except Exception as e:
            wait = 2 ** attempt + random.uniform(0, 1)
            print(f"[RETRY] Attempt {attempt+1}/{max_retries} failed: {e}, "
                  f"waiting {wait:.1f}s")
            time.sleep(wait)
    return None
```

## 四、IP 代理池

### 4.1 免费代理采集

```python
import requests
from lxml import etree
import threading
import queue

# 免费代理源（示例）
FREE_PROXY_SOURCES = [
    'https://www.zdaye.com/free/',
    'https://www.kuaidaili.com/free/',
    'http://www.ip3366.net/free/',
]

def fetch_proxies_from_source(url):
    """从单个源采集代理IP"""
    proxies_list = []
    try:
        resp = requests.get(url, timeout=10)
        tree = etree.HTML(resp.text)
        # 提取 IP:Port（各站 xpath 不同，这里示例）
        for tr in tree.xpath('//table/tbody/tr'):
            ip = tr.xpath('./td[1]/text()')
            port = tr.xpath('./td[2]/text()')
            if ip and port:
                proxies_list.append(f"{ip[0].strip()}:{port[0].strip()}")
    except Exception as e:
        print(f"Failed to fetch from {url}: {e}")
    return proxies_list


def validate_proxy(proxy, test_url='http://httpbin.org/ip', timeout=5):
    """验证代理可用性"""
    try:
        resp = requests.get(test_url,
                           proxies={'http': f'http://{proxy}',
                                    'https': f'http://{proxy}'},
                           timeout=timeout)
        return resp.status_code == 200
    except Exception:
        return False


def build_proxy_pool():
    """采集 + 多线程验证 → 有效代理池"""
    all_proxies = []
    for src in FREE_PROXY_SOURCES:
        all_proxies.extend(fetch_proxies_from_source(src))

    valid_proxies = []
    q = queue.Queue()
    for p in all_proxies:
        q.put(p)

    def worker():
        while not q.empty():
            p = q.get()
            if validate_proxy(p):
                valid_proxies.append(p)
                print(f"[VALID] {p}")
            q.task_done()

    threads = [threading.Thread(target=worker) for _ in range(20)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()

    return valid_proxies
```

### 4.2 代理轮换策略

```python
import itertools

class ProxyRotator:
    def __init__(self, proxy_list):
        self.proxies = proxy_list
        self._cycle = itertools.cycle(proxy_list)  # 广度优先轮换
        self.fail_count = {}  # 记录每个代理的失败次数

    def get_proxy(self):
        return next(self._cycle)

    def mark_failed(self, proxy):
        """标记失败，失败超过3次则移除"""
        self.fail_count[proxy] = self.fail_count.get(proxy, 0) + 1
        if self.fail_count[proxy] >= 3:
            self.proxies.remove(proxy)
            self._cycle = itertools.cycle(self.proxies)
            print(f"[PROXY] Removed {proxy} after 3 failures")

    def get_proxies_dict(self, proxy):
        return {'http': f'http://{proxy}', 'https': f'http://{proxy}'}
```

## 五、字体反爬识别与绕过

### 5.1 原理

网站将关键数字/文字的 Unicode 映射打乱，通过自定义 `.woff`/`.ttf` 字体文件渲染正确字形。爬虫拿到的 HTML 是乱码，但浏览器看到的正常文字。

### 5.2 完整破解流程

```python
import requests
import re
from fontTools.ttLib import TTFont
import io

def crack_font_anti_crawl(html_text, font_base_url):
    """
    破解字体反爬的完整流程
    返回：将乱码字符替换为正确字符后的文本
    """

    # Step 1: 从 HTML 中提取字体文件 URL
    font_url_match = re.search(r'url\("([^"]+\.woff2?)"\)', html_text)
    if not font_url_match:
        font_url_match = re.search(r"url\('([^']+\.woff2?)'\)", html_text)

    if not font_url_match:
        print("[WARN] No font file found in HTML, skipping font cracking")
        return html_text

    font_url = font_url_match.group(1)
    if not font_url.startswith('http'):
        font_url = font_base_url.rstrip('/') + '/' + font_url.lstrip('/')

    print(f"[FONT] Found font URL: {font_url}")

    # Step 2: 下载字体文件
    font_resp = requests.get(font_url, timeout=15)
    if font_resp.status_code != 200:
        print(f"[WARN] Failed to download font: {font_resp.status_code}")
        return html_text

    # Step 3: 解析字体 cmap 表（Unicode → 字形名称）
    font = TTFont(io.BytesIO(font_resp.content))
    font.saveXML("debug_font.xml")  # 调试用：导出字体 XML

    # 获取 cmap 表：Unicode 码点 → 字形名称
    cmap = font.getBestCmap()  # {unicode_int: glyph_name}
    print(f"[FONT] cmap has {len(cmap)} mappings")

    # Step 4: 获取字形顺序表（glyphOrder）
    # 标准字体中 glyphOrder[2:] 对应数字 0-9
    glyph_order = font.getGlyphOrder()

    # Step 5: 构建映射：乱码 Unicode → 正确数字/字符
    # 策略：对比当前字体与标准字体的 glyph 轮廓相似度
    #      或直接用 glyphOrder 索引位置映射

    # 简化方案：假设 glyph 顺序未变，只有 Unicode 映射变了
    # 构建 Unicode → 正确字符的映射
    decode_map = {}
    standard_glyph_to_char = _get_standard_glyph_map(font)  # 需要预先建立基准

    for unicode_int, glyph_name in cmap.items():
        if glyph_name in standard_glyph_to_char:
            decode_map[chr(unicode_int)] = standard_glyph_to_char[glyph_name]

    print(f"[FONT] Built decode map with {len(decode_map)} entries")

    # Step 6: 替换 HTML 中的乱码字符为正确字符
    decoded_text = html_text
    for scrambled_char, real_char in decode_map.items():
        decoded_text = decoded_text.replace(scrambled_char, real_char)

    return decoded_text


def _get_standard_glyph_map(font):
    """
    建立基准映射：字形名称 → 正确字符
    方法：下载同一网站的一个标准字体文件作为基准，
         或使用 glyph 轮廓向量比较（更鲁棒）
    """
    # 实际使用时，可以预先手动建立一个基准映射表
    # 这里以某常见字体为例（需要根据目标网站调整）
    glyph_order = font.getGlyphOrder()

    # 常见规律：glyphOrder 前几个是 .notdef, .null, 然后是数字0-9
    standard_map = {}
    # 示例：假设 glyphOrder[2:12] 是按顺序的数字 0-9
    char_sequence = list("0123456789")
    # 实际使用时需要根据字体具体情况调整
    return standard_map
```

### 5.3 更鲁棒的方案：glyph 轮廓比对

```python
def compare_glyphs_by_outline(font, baseline_font):
    """
    通过比对 glyph 轮廓坐标来精确匹配
    这比依赖 glyphOrder 更可靠，因为不同字体文件 glyph 顺序可能完全不同
    """
    from fontTools.pens.pointPen import PointToSegmentPen

    def get_glyph_coords(f, glyph_name):
        """提取字形轮廓坐标"""
        glyph = f.getGlyphSet()[glyph_name]
        pen = PointToSegmentPen()
        glyph.draw(pen)
        return pen.path

    # 对每个未知字体的 glyph，找基准字体中轮廓最相似的
    result_map = {}
    unknown_cmap = font.getBestCmap()
    baseline_cmap = baseline_font.getBestCmap()

    for u_int, g_name in unknown_cmap.items():
        u_coords = get_glyph_coords(font, g_name)
        best_match = None
        best_score = float('inf')

        for b_u, b_g in baseline_cmap.items():
            b_coords = get_glyph_coords(baseline_font, b_g)
            # 计算坐标相似度（简化：比较坐标点数量差）
            score = abs(len(str(u_coords)) - len(str(b_coords)))
            if score < best_score:
                best_score = score
                best_match = chr(b_u)

        result_map[chr(u_int)] = best_match

    return result_map
```

## 六、CSS 偏移反爬识别与绕过

### 6.1 原理

文字被拆成单个 `<span>`，通过 CSS `left` 属性打乱 DOM 顺序，视觉正确但爬取乱序。

```html
<div class="name">
  <span class="char" style="left:30px">股</span>
  <span class="char" style="left:0px">贵</span>
  <span class="char" style="left:15px">台</span>
</div>
<!-- 视觉顺序：贵州茅台，DOM顺序：股台贵台 -->
```

### 6.2 破解代码

```python
import re
from selenium import webdriver
from selenium.webdriver.common.by import By

def parse_css_offset_text(driver, css_selector):
    """
    解析被 CSS 偏移打乱的文本
    使用 Selenium 获取每个字符的 left 像素值，按从左到右排序还原
    """
    elements = driver.find_elements(By.CSS_SELECTOR, f"{css_selector} .char")
    if not elements:
        return None

    chars_with_pos = []
    for el in elements:
        text = el.text.strip()
        style = el.get_attribute('style') or ''
        # 提取 left 值
        left_match = re.search(r'left:\s*([-\d]+)px', style)
        left_val = int(left_match.group(1)) if left_match else 0
        chars_with_pos.append({'text': text, 'left': left_val})

    # 按 left 值从小到大排序（还原从左到右的阅读顺序）
    chars_with_pos.sort(key=lambda x: x['left'])
    return ''.join(c['text'] for c in chars_with_pos)


def parse_css_offset_from_html(html_text):
    """
    无浏览器方案：从 HTML 源码直接解析 CSS 偏移
    适用于静态渲染的场景（服务端返回的 HTML 已包含 style 属性）
    """
    import re
    # 匹配所有 .char 元素及其 style 中的 left 值
    pattern = r'<span[^>]*class="[^"]*char[^"]*"[^>]*style="[^"]*left:\s*([-\d]+)px[^"]*"[^>]*>([^<]*)</span>'
    matches = re.findall(pattern, html_text)

    if not matches:
        return None

    # 按 left 值排序
    matches.sort(key=lambda x: int(x[0]))
    return ''.join(m[1] for m in matches)
```

## 九、完整爬虫模板

```python
"""
反反爬爬虫标准模板
使用前：pip install requests fake_useragent fonttools lxml
"""
import requests
import time
import random
from fake_useragent import UserAgent
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

class AntiCrawlSpider:
    def __init__(self, name="default"):
        self.name = name
        self.ua = UserAgent()
        self.session = self._build_session()
        self.proxy_list = []
        self.fail_count = 0

    def _build_session(self):
        session = requests.Session()
        retry = Retry(total=3, backoff_factor=1.5,
                     status_forcelist=[429, 500, 502, 503, 504])
        session.mount("http://", HTTPAdapter(max_retries=retry))
        session.mount("https://", HTTPAdapter(max_retries=retry))
        return session

    def get_headers(self, referer=None):
        h = {
            'User-Agent': self.ua.random,
            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
            'Accept-Language': 'zh-CN,zh;q=0.9,en;q=0.8',
            'Connection': 'keep-alive',
        }
        if referer:
            h['Referer'] = referer
        return h

    def get(self, url, referer=None, max_retries=3):
        for attempt in range(max_retries):
            try:
                time.sleep(random.uniform(1, 3))
                kwargs = {'timeout': 30}
                if self.proxy_list:
                    kwargs['proxies'] = self._random_proxy()
                resp = self.session.get(url, headers=self.get_headers(referer), **kwargs)

                if resp.status_code == 403:
                    print(f"[WARN] 403 on {url}, switching data source")
                    return None
                if resp.status_code == 200:
                    return resp
            except Exception as e:
                wait = 2 ** attempt + random.uniform(0, 1)
                print(f"[RETRY] {url}: attempt {attempt+1}, waiting {wait:.1f}s: {e}")
                time.sleep(wait)

        print(f"[FAIL] Failed after {max_retries} retries: {url}")
        return None

    def _random_proxy(self):
        proxy = random.choice(self.proxy_list)
        return {'http': f'http://{proxy}', 'https': f'http://{proxy}'}
```
