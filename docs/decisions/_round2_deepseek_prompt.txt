2. **A2 utils/calendar.py + exit_advisor.py** 日历日改交易日
```python
# utils/calendar.py
def count_trading_days(start_date, end_date, data_dir=None) -> int:
    """统计 [start_date, end_date] 区间的交易日数（包含两端，闭区间）。
    数据源优先级：
    1. data/stock_YYYYMMDD.csv 文件名
    2. 兜底：用 weekday < 5 当作交易日
    """
    s, e = _to_date(start_date), _to_date(end_date)
    if s is None or e is None or s > e:
        return 0
    local_days = _list_local_trading_days(data_dir)
    if local_days:
        s_str, e_str = s.strftime('%Y-%m-%d'), e.strftime('%Y-%m-%d')
        in_range = [d for d in local_days if s_str <= d <= e_str]
        if in_range:
            return len(in_range)
    # fallback weekday
    n = 0
    cur = s
    while cur <= e:
        if cur.weekday() < 5:
            n += 1
        cur += timedelta(days=1)
    return n

# exit_advisor.py

    except Exception:
        hold_days = 0
else:
    hold_days = 0
```
